# Format: TC###: param1=value1&param2=value2

FA001: type=seasonal&color=yellow
FA002: type=tropical&color=orange
FA003: type=seasonal&color=red
FA004: type=exotic&color=green
FA005: type=seasonal&color=yellow&organic=true

FA007: type=&color=
